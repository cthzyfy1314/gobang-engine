# Day 5-7 (5/3-5/5): Forbidden Move Detection (Renju 黑专属禁手)

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:executing-plans.

**Goal:** 实现国规黑方三种禁手判定（长连 / 四四 / 三三），集成到 search 让黑方 αβ 避免下禁手位置。**国规 9.2-c 边界规则**：黑五连 + 禁手同时形成 → 五连优先（禁手失效）。

- search 集成：生成黑方候选 move 时跳过禁手位置，但保留五连位置。

### Task A2: forbid.c 起手 + 11 长度居中线提取 + 五连/长连识别

**Files:** Create `src/forbid.c`

提取 4 方向居中线 + 在中心点（line[5]）落黑后，扫连续段判 five / overline。

```c
/* 在 (cr, cc) 沿方向 (dr, dc) 提取 11 长度居中线（5 左 + 中心 + 5 右）
 * 越界格用 -1 标记（边界），其他用 cells 值。
 */
static void extract_centered(const Board *b, int cr, int cc, int dr, int dc, int8_t line[11]) {
    for (int k = -5; k <= 5; k++) {
        int r = cr + k * dr, c = cc + k * dc;
        if (board_in_bounds(r, c)) line[k + 5] = (int8_t)b->cells[r][c];
        else                        line[k + 5] = -1;
    }
}

/* 在中心点（line[5]）位置假定为 BLACK，分析包含 center 的棋型 */
static int center_run_length(const int8_t *line) {
    /* 假定 line[5] = BLACK，向左右扩展统计连续 BLACK 段长度 */
    int len = 1;
    for (int i = 4; i >= 0; i--) { if (line[i] == 1) len++; else break; }
    for (int i = 6; i <= 10; i++) { if (line[i] == 1) len++; else break; }
    return len;
}
```

在 `forbid_check_black` 主函数中：
1. 将中心点临时设为 BLACK（拷贝 board 或用 const_cast 模式但不修改 b）
2. 4 方向各调 extract_centered + center_run_length
3. 若有 length == 5 → five，立即返回 FORBID_NONE（五连优先）
4. 若有 length >= 6 → overline 累加
5. 4 方向后：overline >= 1 → FORBID_OVERLINE

### Task A3: 冲四 / 活四识别（含跳冲四）

5 种"四"形式（在中心线 line[0..10] 中查找包含 center=line[5] 的）：
- `XXXX_` 或 `_XXXX`（连续 4 + 一端空，冲四）
- `XXX_X` / `X_XXX`（跳冲四）
- `XX_XX`（中心间隔的跳冲四）
- `_XXXX_`（连续 4 + 双端空，活四）

为简化匹配，**用 5 长度子串遍历**：
```c
/* 数 line 上有多少个"四"棋型（含活四 + 冲四）经过 center 点 */
static int count_fours_through_center(const int8_t *line, bool *open_four_found) {
    /* 遍历所有 5 长度子串 line[i..i+4]，i in [1, 5] 让窗口包含 center=5 */
    int four_count = 0;
    *open_four_found = false;

    for (int i = 1; i <= 5; i++) {
        int blacks = 0, empties = 0;
        for (int k = 0; k < 5; k++) {
            int v = line[i + k];
            if (v == 1) blacks++;
            else if (v == 0) empties++;
            else { blacks = -1; break; }
        }
        if (blacks == 4 && empties == 1) {
            /* 是一个"四"形式。检查活四：左右各扩展 1 格都是 EMPTY */
            int left = (i - 1 >= 0) ? line[i - 1] : -1;
            int right = (i + 5 <= 10) ? line[i + 5] : -1;
            if (left == 0 && right == 0) {
                /* 活四：但要确认 4 子是 _XXXX_（中间无 gap）—— 否则只是 _XXX_X_ 这种跳四中的 4 子，不是活四 */
                /* 简化：算作活四+冲四混合，记 1 个四 */
                if (line[i] == 1 && line[i+1] == 1 && line[i+2] == 1 && line[i+3] == 1 && line[i+4] == 1) {
                    /* 实际上 blacks==5，已经被排除上面 if 。这里不会到达 */
                }
                /* 只有 _XXXX_（连续）才算活四 */
                int run_idx = -1, run_len = 0;
                for (int k = 0; k < 5; k++) {
                    if (line[i + k] == 1) {
                        if (run_idx == -1) run_idx = k;
                        run_len++;
                    }
                }
                if (run_len == 4 && line[i + run_idx - 1] == 0 /* impossible since blacks==4 */) {
                    /* skip */
                }
                /* 因为 blacks==4 && empties==1，5 长度子串内有 1 个空隙。
                 * 若空隙在边缘 (k=0 或 4)，则 4 子连续 → 看左/右扩展决定活四/冲四
                 * 若空隙在中间 (k=1,2,3)，则是跳冲四，必为冲四（不可能活四）
                 */
                int empty_pos = -1;
                for (int k = 0; k < 5; k++) if (line[i + k] == 0) { empty_pos = k; break; }
                if (empty_pos == 0 || empty_pos == 4) {
                    /* 4 子连续。empty 在端 → 活四需另一端也开放 */
                    int run_left = (empty_pos == 0) ? i + 1 : i;     /* 4 子的最左 index */
                    int run_right = run_left + 3;                    /* 4 子的最右 index */
                    int outer_left  = (run_left  - 1 >= 0)  ? line[run_left - 1]  : -1;
                    int outer_right = (run_right + 1 <= 10) ? line[run_right + 1] : -1;
                    if (outer_left == 0 && outer_right == 0) {
                        *open_four_found = true;
                        four_count++;
                        return four_count; /* 活四是绝对优势，立即返回 */
                    } else {
                        four_count++;
                    }
                } else {
                    /* 跳冲四 */
                    four_count++;
                }
            } else if (left == 0 || right == 0) {
                /* 一端开放：冲四（连续 4 一端被堵或边界） */
                four_count++;
            }
            /* else 双端被堵 → 不是有效四 */
        }
    }
    return four_count;
}
```

注：此函数会重复计数同一个跳四。**简化**：返回值 ≥ 1 表示"至少有四"，>= 2 表示"双四"。

为了避免重复计数，我们改成：每个方向最多记 1 个"四"（任意类型），4 个方向总和 ≥ 2 = 双四。这是合理简化。

### Task A4: forbid_check_black 主函数 + 长连 + 四四判定

```c
ForbidType forbid_check_black(const Board *b, int row, int col) {
    if (!board_in_bounds(row, col)) return FORBID_NONE;
    if (b->cells[row][col] != EMPTY) return FORBID_NONE;

    /* 临时落黑（用栈拷贝避免修改 b） */
    Board copy = *b;
    copy.cells[row][col] = BLACK;

    static const int dirs[4][2] = { {0,1}, {1,0}, {1,1}, {1,-1} };

    int five_count = 0;
    int overline_count = 0;
    int four_count_dirs = 0;   /* 4 方向中有"至少一个四"的方向数 */
    /* three_count: 5/4 实现 */

    for (int d = 0; d < 4; d++) {
        int8_t line[11];
        extract_centered(&copy, row, col, dirs[d][0], dirs[d][1], line);
        int run = center_run_length(line);
        if (run == 5) five_count++;
        else if (run >= 6) overline_count++;
        else {
            bool unused;
            int n_fours = count_fours_through_center(line, &unused);
            if (n_fours >= 1) four_count_dirs++;
        }
    }

    /* 国规 9.2-c：五连优先 */
    if (five_count > 0) return FORBID_NONE;

    if (overline_count > 0) return FORBID_OVERLINE;
    if (four_count_dirs >= 2) return FORBID_DOUBLE_FOUR;

    /* 5/4 阶段加入 three_count */
    return FORBID_NONE;
}
```

### Task B1: 活三 + 跳活三识别

在中心线 line[0..10] 中找包含 center=line[5] 的"形式上活三"：

```c
/* 数 line 上有多少个"形式上活三"经过 center */
static int count_open_threes_through_center(const int8_t *line) {
    int count = 0;
    /* 形式 1: _XXX_ 连续活三（6 长度窗口） */
    for (int i = 0; i <= 5; i++) {
        if (i + 5 > 10) continue;
        if (line[i] == 0 && line[i+1] == 1 && line[i+2] == 1 && line[i+3] == 1
            && line[i+4] == 0 && line[i+5] != 1) {
            /* 包含 center? center index = 5, 黑子在 i+1..i+3 */
            if (5 >= i+1 && 5 <= i+3) {
                /* 进一步：再左扩展 line[i-1] != 1（避免被识别为冲四的一部分） */
                int left2 = (i - 1 >= 0) ? line[i - 1] : -1;
                if (left2 != 1) count++;
            }
        }
    }
    /* 形式 2: _XX_X_ 跳活三（6 长度窗口） */
    for (int i = 0; i <= 4; i++) {
        if (line[i] == 0 && line[i+1] == 1 && line[i+2] == 1
            && line[i+3] == 0 && line[i+4] == 1 && line[i+5] == 0) {
            if (5 >= i+1 && 5 <= i+4) count++;
        }
    }
    /* 形式 3: _X_XX_ 跳活三（对称） */
    for (int i = 0; i <= 4; i++) {
        if (line[i] == 0 && line[i+1] == 1 && line[i+2] == 0
            && line[i+3] == 1 && line[i+4] == 1 && line[i+5] == 0) {
            if (5 >= i+1 && 5 <= i+4) count++;
        }
    }
    return count;
}
```
